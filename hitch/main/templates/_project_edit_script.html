const settingsProjectSelect = document.querySelector("[data-settings-project-select]"); const projectEditOpen = document.querySelector("[data-project-edit-open]"); const projectEditDialog = document.querySelector("[data-project-edit-dialog]"); const projectEditCancel = projectEditDialog && projectEditDialog.querySelector("[data-project-edit-cancel]"); const projectEditId = projectEditDialog && projectEditDialog.querySelector("[data-project-edit-id]"); const projectEditName = projectEditDialog && projectEditDialog.querySelector("[data-project-edit-name]"); const projectEditRepo = projectEditDialog && projectEditDialog.querySelector("[data-project-edit-repo]"); const projectEditExtraSystemPrompt = projectEditDialog && projectEditDialog.querySelector("[data-project-edit-extra-system-prompt]"); const projectEditAutoPrMode = projectEditDialog && projectEditDialog.querySelector("[data-project-edit-auto-pr-mode]"); function selectedProjectOption() { if (!settingsProjectSelect) return null; const option = settingsProjectSelect.selectedOptions && settingsProjectSelect.selectedOptions[0]; return option && option.value ? option : null; } function syncProjectEditButton() { if (projectEditOpen) projectEditOpen.disabled = !selectedProjectOption(); } function populateProjectEditDialog() { const option = selectedProjectOption(); if (!option) return false; if (projectEditId) projectEditId.value = option.value; if (projectEditName) projectEditName.value = option.dataset.projectName || ""; if (projectEditRepo) projectEditRepo.value = option.dataset.projectRepoPath || ""; if (projectEditExtraSystemPrompt) projectEditExtraSystemPrompt.value = option.dataset.projectExtraSystemPrompt || ""; if (projectEditAutoPrMode) { projectEditAutoPrMode.value = option.dataset.projectAutoPrMode || "follow_global"; } return true; } if (settingsProjectSelect) settingsProjectSelect.addEventListener("change", syncProjectEditButton); syncProjectEditButton(); if (projectEditOpen) { projectEditOpen.addEventListener("click", () => { if (!populateProjectEditDialog()) return; showDialog(projectEditDialog); if (projectEditName) projectEditName.focus(); }); } if (projectEditCancel) projectEditCancel.addEventListener("click", () => hideDialog(projectEditDialog));
