function initializeUsage() { document.querySelectorAll("[data-usage-chart-toggle]").forEach((toggle) => { const chart = document.getElementById(toggle.getAttribute("aria-controls")); if (!chart) return; toggle.addEventListener("click", () => { const expanded = chart.classList.toggle("is-expanded"); toggle.setAttribute("aria-expanded", expanded ? "true" : "false"); }); }); for (const el of document.querySelectorAll("time[data-quota-checked-at]")) { const date = new Date(el.dateTime); if (!Number.isFinite(date.getTime())) continue; el.textContent = date.toLocaleString(undefined, { year: "numeric", month: "short", day: "numeric", hour: "numeric", minute: "2-digit", second: "2-digit", timeZoneName: "short", }); } for (const el of document.querySelectorAll("time[data-resets-at]")) { const seconds = Number(el.dataset.resetsAt); if (!Number.isFinite(seconds) || seconds <= 0) continue; const date = new Date(seconds * 1000); el.dateTime = date.toISOString(); el.title = date.toString(); el.textContent = window.hitch.relativeFromNow(date); } } initializeUsage(); let usageTimer = 0; let usageRefreshing = false; let usageFailures = 0; async function refreshUsage() { window.clearTimeout(usageTimer); if (usageRefreshing) return; let delay = 1000; try { if (document.hidden) return; const root = document.querySelector("[data-usage-root]"); if (!root) return; usageRefreshing = true; const url = new URL(root.dataset.refreshUrl, window.location.href); url.searchParams.set("cursor", root.dataset.cursor); url.searchParams.set("profile", root.dataset.profile); const response = await fetch(url, { cache: "no-store", signal: AbortSignal.timeout(15000) }); if (!response.ok) throw new Error("Usage refresh failed"); const result = await response.json(); const parsed = new DOMParser().parseFromString(result.html, "text/html"); const next = parsed.querySelector("[data-usage-root]"); if (!next) throw new Error("Usage response missing"); // Carry chart expansion through the refreshed markup. const focusedChartId = root.contains(document.activeElement) ? document.activeElement.getAttribute("aria-controls") : null; next.querySelectorAll("[data-usage-chart-toggle]").forEach((toggle) => { const chartId = toggle.getAttribute("aria-controls"); const previous = root.querySelector(`[aria-controls="${chartId}"]`); const chart = next.querySelector(`#${chartId}`); if (previous?.getAttribute("aria-expanded") === "true" && chart) { toggle.setAttribute("aria-expanded", "true"); chart.classList.add("is-expanded"); } }); root.replaceWith(next); initializeUsage(); if (focusedChartId) next.querySelector(`[aria-controls="${focusedChartId}"]`)?.focus(); usageFailures = 0; // Token polling ends with the sweep; quota still gets its // normal two-minute refresh while the page remains open. delay = result.should_poll ? 1000 : 120000; } catch (error) { usageFailures += 1; delay = Math.min(30000, 1000 * 2 ** usageFailures); if (usageFailures >= 3) { const root = document.querySelector("[data-usage-root]"); let note = root && root.querySelector("[data-usage-error]"); if (root && !note) { note = document.createElement("p"); note.dataset.usageError = "true"; note.setAttribute("role", "status"); root.prepend(note); } if (note) note.textContent = "Unable to refresh usage. Showing the last available values; retrying shortly."; } } finally { usageRefreshing = false; usageTimer = window.setTimeout(refreshUsage, delay); } } const usageRoot = document.querySelector("[data-usage-root]"); if (usageRoot) usageTimer = window.setTimeout(refreshUsage, usageRoot.dataset.shouldPoll === "true" ? 1000 : 120000); document.addEventListener("visibilitychange", () => { if (!document.hidden) refreshUsage(); }); window.addEventListener("pageshow", (event) => { if (event.persisted) refreshUsage(); });
