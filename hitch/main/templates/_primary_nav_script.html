const navMenu = document.querySelector("[data-nav-menu]"); const navMenuButton = navMenu && navMenu.querySelector("[data-nav-menu-open]"); function closeNavMenu() { if (!navMenuButton) return; navMenuButton.setAttribute("aria-expanded", "false"); } if (navMenuButton) { navMenuButton.addEventListener("click", (event) => { event.preventDefault(); const willOpen = navMenuButton.getAttribute("aria-expanded") !== "true"; navMenuButton.setAttribute("aria-expanded", willOpen ? "true" : "false"); }); document.addEventListener("click", (event) => { if (!navMenu || navMenu.contains(event.target)) return; closeNavMenu(); }); document.addEventListener("keydown", (event) => { if (event.key !== "Escape" || navMenuButton.getAttribute("aria-expanded") !== "true") return; closeNavMenu(); navMenuButton.focus(); }); if (navMenu) navMenu.classList.add("primary-nav-js"); }
